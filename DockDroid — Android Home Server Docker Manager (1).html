import os, time, psutil, docker, uvicorn
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"] )
client = docker.from_env()
@app.get("/api/status")
def status():
    m, d = psutil.virtual_memory(), psutil.disk_usage("/")
    cs = client.containers.list(all=True)
    return {"hostname": os.uname().nodename, "cpu_percent": psutil.cpu_percent(0.1), "ram_used_gb": round(m.used/1e9,1), "ram_total_gb": round(m.total/1e9,1), "ram_percent": m.percent, "disk_used_gb": round(d.used/1e9,0), "disk_total_gb": round(d.total/1e9,0), "disk_percent": d.percent, "containers_total": len(cs), "containers_running": sum(1 for c in cs if c.status=="running")}
@app.get("/api/containers")
def containers(check_updates: bool = False):
    out = []
    for c in client.containers.list(all=True):
        img = (c.image.tags or [c.attrs.get("Config",{}).get("Image","unknown")])[0]
        out.append({"id": c.short_id, "name": c.name, "image": img, "status": c.status, "ports": list((c.attrs.get("NetworkSettings",{}).get("Ports") or {}).keys()), "update_available": False})
    return {"containers": out}
